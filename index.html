Integrasi Google Sheet

Untuk menyambungkan form ini (beserta fitur upload fotonya) ke Google Sheet Anda, perbarui script di Apps Script menjadi kode berikut:

  1. Buka Google Sheet "SPKL Padalarang".
  2. Klik menu Ekstensi > Apps Script.
  3. Hapus kode lama, ganti dengan kode di bawah. Simpan.
  4. Penting: Klik Terapkan (Deploy) > Deployment Baru (Pilih Aplikasi Web, Akses Siapa Saja). Copy URL baru tersebut dan paste di bawah ini.
function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);

    // Coba cari sheet dengan nama "Rekap Lembur", jika tidak ada pakai sheet aktif
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("Rekap Lembur");
    if (!sheet) {
      // Fallback: pakai sheet pertama / sheet aktif
      sheet = ss.getActiveSheet();
    }

    var fileUrl = "";

    // Upload foto ke Google Drive jika ada
    if (data.fotoBase64 && data.fotoBase64 !== "") {
      var byteCharacters = Utilities.base64Decode(data.fotoBase64);
      var blob = Utilities.newBlob(byteCharacters, data.fotoMimeType, data.fotoFileName);
      var file = DriveApp.createFile(blob);
      file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      fileUrl = file.getUrl();
    }

    // Cari baris kosong pertama mulai dari baris 4 (A4) karena baris 1-3 di-merge
    var lastRow = sheet.getLastRow();
    var nextRow = lastRow < 4 ? 4 : lastRow + 1;

    // Tulis data ke kolom A-O sesuai header di baris 3
    // A=NO INDUK, B=NAMA, C=JABATAN, D=PENEMPATAN,
    // E=KATEGORI, F=KETERANGAN LEMBUR, G=UPAH,
    // H=TGL LEMBUR, I=SHIFT, J=JAM KERJA,
    // K=WAKTU LEMBUR/KJK, L=Jml JAM, M=KETERANGAN, N=EVIDEN, O=MINGGU KE
    sheet.getRange(nextRow, 1, 1, 15).setValues([[
      data.noInduk || "",
      data.nama || "",
      data.jabatan || "",
      data.penempatan || "",
      data.kategori || "",
      data.keteranganLembur || "",
      data.upah || "",
      data.tglLembur || "",
      data.shift || "",
      data.jamKerja || "",
      data.waktuLembur || "",
      data.jmlJam || "",
      data.keterangan || data.keteranganTambahan || "",
      fileUrl,
      data.mingguKe || ""
    ]]);

    return ContentService.createTextOutput(JSON.stringify({"status": "success", "url": fileUrl}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": error.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("petugas");
    if (!sheet) {
      return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": "Sheet 'petugas' tidak ditemukan"}))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    // Baris 1 adalah header (NIP, Password, Nama, Jabatan, Penempatan, Role)
    var data = sheet.getDataRange().getValues();
    var petugas = [];
    
    for (var i = 1; i < data.length; i++) {
      if (data[i][2]) { // Cek jika Nama (Kolom C) tidak kosong
        petugas.push({
          noInduk: data[i][0] || "",
          password: data[i][1] || "",
          nama: data[i][2] || "",
          jabatan: data[i][3] || "",
          penempatan: data[i][4] || "",
          role: data[i][5] || ""
        });
      }
    }
    
    return ContentService.createTextOutput(JSON.stringify({"status": "success", "data": petugas}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": error.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
